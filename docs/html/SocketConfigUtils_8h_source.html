55 // Offset index to account for fixed socket occupying slot 0

57 config.has_fixed_socket ? socket_index - 1 : socket_index;
58 const size_t rot_count = config.has_fixed_socket
59 ? static_cast<size_t>(config.socket_count) - 1
60 : static_cast<size_t>(config.socket_count);
61
62 float base_angle_deg;
63 if (rot_count <= 1) {
64 base_angle_deg = (config.arc_min + config.arc_max) / 2.f;
65 } else {
66 const float span = config.arc_max - config.arc_min;
67 // Full circle: avoid duplicating the start/end position
68 if (std::abs(span - 360.f) < 1e-3f) {
69 base_angle_deg = config.arc_min +

71 static_cast<float>(rot_count);
