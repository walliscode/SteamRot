186 MachinaArchetype Build(std::string archetype_name) {
187 auto steps = m_steps;
188 return MachinaArchetype{
189 archetype_name,
190 [steps = std::move(steps), archetype_name = archetype_name](
191 const PartGraph &parts,
192 uint32_t start_id,
193 uint32_t depth) -> MachinaArchetypeResult {
194 // create MachinaArchetypeResult to accumulate the final result and
195 // trace
196 MachinaArchetypeResult result{false, T{}};
197
198 // create context to hold the trace and any other state we want to
199 // pass
200 ArchetypeAnalysisContext context{};
201
202 // Check for empty PartGraph
203 if (parts.empty()) {
204
205 // this is automatically a failed result
206 result.m_result = false;
207
208 // set up EmtpyGraph event in the trace
209 add_empty_part_graph_event(context);
210 result.m_trace = std::move(context.trace);
211
212 // return early since there's no graph to traverse
213 return result;
214 }
215
216 // Check for empty steps
217 if (steps.empty()) {
218
219 // this is automatically a failed result since a chain with no
220 // steps can't be satisfied
221 result.m_result = false;
222
223 // set up EmptyChainSteps event in the trace to explain why the
224 // result is false
225 add_empty_chain_steps_event(context);
226 result.m_trace = std::move(context.trace);
227
228 // return early since there's no steps to evaluate
229 return result;
230 }
231
232 add_scope_begin_event(context, archetype_name,
233 ScopeKind::MachinaArchetype, parts, depth,
234 start_id);
235
236 // create cursor
237 uint32_t graph_cursor = start_id;
238 // set starting result to true
239 result.m_result = true;
240
241 // cycle through the steps and evalute each one depending on its
242 // kind
243 for (const auto &step : steps) {
244 switch (step.kind) {
245 case ArchetypeStepKind::Sequence: {
246
247 // evalute the current node with the step's descriptor
248 ChainDescriptorResult step_result =
249 step.descriptor(parts, graph_cursor, depth + 1);
250
251 // for a simple sequence step, if the result is false then the
252 // whole archetype fails and we can break early; if it's true then
253 // we can continue to the next step
254 if (!step_result)
255 result.m_result = false;
256
257 // append the step's trace to the overall trace
258 Merge(context.trace, std::move(step_result.m_trace));
259 break;
260 }
261 case ArchetypeStepKind::AtLeastNOf: {
262 break;
263 }
264 }
265 // if result is false then break early without evaluating further
266 // steps
267 if (!result.m_result)
268 break;
269 }
270 add_scope_end_event(context, archetype_name, ScopeKind::MachinaArchetype,
271 result.m_result, depth);
272 // scope events complete; move trace and return result
273 result.m_trace = std::move(context.trace);
274 return result;
275 }};
276 }
