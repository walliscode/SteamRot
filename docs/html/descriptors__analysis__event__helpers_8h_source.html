25inline std::string resolve_part_alias(const PartGraph &parts, uint32_t id) {
26 if (const auto part_it = parts.find(id); part_it != parts.end()) {
27 return std::visit(
28 [](const auto &inst) -> const std::string & { return inst.alias; },
29 part_it->second);
30 }
31 return {};
32}
